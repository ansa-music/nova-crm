========== firestore.rules ==========

rules_version = '2';

// ---------------------------------------------------------------------------
// Access model: the workspace Owner has blanket access to everything. Every
// other member — regardless of role — can only read/write a page (and its
// rows) if their uid is explicitly listed in that page's `allowedUsers`
// array. This is the actual security boundary; the React app's own checks
// (utils/permissions.ts) exist only for UX and must never be relied on alone.
// ---------------------------------------------------------------------------

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function memberDoc(workspaceId, uid) {
      return get(/databases/$(database)/documents/workspaces/$(workspaceId)/members/$(uid));
    }

    function isMember(workspaceId) {
      return signedIn() &&
        exists(/databases/$(database)/documents/workspaces/$(workspaceId)/members/$(request.auth.uid));
    }

    function myRole(workspaceId) {
      return memberDoc(workspaceId, request.auth.uid).data.role;
    }

    function isOwner(workspaceId) {
      return isMember(workspaceId) && myRole(workspaceId) == 'owner';
    }

    function isOwnerOrAdmin(workspaceId) {
      return isMember(workspaceId) && (myRole(workspaceId) == 'owner' || myRole(workspaceId) == 'admin');
    }

    // Owner sees/manages every page unconditionally. Anyone else needs their
    // uid explicitly present in the page's allowedUsers list.
    function canAccessPage(workspaceId, pageData) {
      return isOwner(workspaceId) ||
        (isMember(workspaceId) && request.auth.uid in pageData.get('allowedUsers', []));
    }

    // Viewers can open a page but never write row data on it.
    // Being in allowedUsers only grants VIEW access now. Editing row data
    // (or managing subpages) requires being explicitly listed in
    // editableUsers too — a separate grant the Owner or this page's
    // responsible person controls independently of plain view access.
    function canEditPage(workspaceId, pageData) {
      return isOwner(workspaceId) ||
        pageData.get('responsibleUserId', null) == request.auth.uid ||
        (canAccessPage(workspaceId, pageData) && request.auth.uid in pageData.get('editableUsers', []));
    }

    // ---------------------------------------------------------------------
    // Users
    // ---------------------------------------------------------------------
    match /users/{uid} {
      allow read: if signedIn();
      allow write: if signedIn() && request.auth.uid == uid;
    }

    // ---------------------------------------------------------------------
    // Workspaces
    // ---------------------------------------------------------------------
    match /workspaces/{workspaceId} {
      // Any signed-in user may read the workspace's own top-level doc (name,
      // icon, color, ownerId — nothing sensitive) so the public /join/:id
      // page can show "Request access to «Name»" before they're a member.
      // Everything under it (members, pages, rows, history) stays gated by
      // isMember()/isOwner() as before.
      allow read: if signedIn();
      // Only this one account may create new workspaces. Everyone else
      // must be invited or approved via a /join/:workspaceId request — see
      // the joinRequests block below. Keep in sync with
      // src/utils/adminAccess.ts (WORKSPACE_ADMIN_EMAIL) on the client.
      allow create: if signedIn()
        && request.resource.data.ownerId == request.auth.uid
        && request.auth.token.email == 'nurpro2005@gmail.com';
      allow update: if isOwner(workspaceId);
      allow delete: if isOwner(workspaceId);

      // -----------------------------------------------------------------
      // Members — only the Owner manages membership, roles, and removal.
      // -----------------------------------------------------------------
      match /members/{memberId} {
        // Existing members can read the full roster. A signed-in user who
        // is NOT yet a member may still read the one pending-invite doc
        // that matches their own email — otherwise claimPendingInvites()
        // could never find its own invite to accept (chicken-and-egg).
        allow read: if isMember(workspaceId) ||
          (signedIn() && memberId == request.auth.token.email && resource.data.status == 'invited');

        // Self-service: a member may update just their own nickname and/or
        // presence heartbeat on their own member doc, without needing
        // Owner-level member-management rights for anything else.
        allow update: if signedIn()
          && memberId == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['nickname', 'lastActiveAt']);

        // Owner manages any membership record: invite, change role, remove.
        allow create, update, delete: if isOwner(workspaceId);

        // Bootstrap: the workspace creator may create their own Owner record.
        allow create: if signedIn()
          && request.auth.uid == memberId
          && request.resource.data.role == 'owner'
          && get(/databases/$(database)/documents/workspaces/$(workspaceId)).data.ownerId == request.auth.uid;

        // Invite acceptance: a user may create their own uid-keyed record
        // only if a matching PENDING invite (keyed by their lowercase email)
        // exists, and role/email must match exactly what was invited.
        allow create: if signedIn()
          && request.auth.uid == memberId
          && request.auth.token.email == request.resource.data.email
          && request.resource.data.status == 'active'
          && exists(/databases/$(database)/documents/workspaces/$(workspaceId)/members/$(request.resource.data.email))
          && memberDoc(workspaceId, request.resource.data.email).data.status == 'invited'
          && memberDoc(workspaceId, request.resource.data.email).data.role == request.resource.data.role;

        // Cleanup of the placeholder email-keyed invite doc once claimed.
        allow delete: if signedIn()
          && memberId == resource.data.email
          && resource.data.status == 'invited'
          && isMember(workspaceId)
          && memberDoc(workspaceId, request.auth.uid).data.email == resource.data.email;
      }

      // -----------------------------------------------------------------
      // Join requests — the "request to join instead of creating your own
      // workspace" flow behind a shared /join/:workspaceId link. Anyone
      // signed in may create exactly one request doc for themselves; only
      // the Owner may read the list, approve (by creating a real member
      // record separately), or reject.
      // -----------------------------------------------------------------
      match /joinRequests/{uid} {
        allow read: if isOwner(workspaceId) || (signedIn() && request.auth.uid == uid);
        allow create: if signedIn()
          && request.auth.uid == uid
          && request.resource.data.uid == uid
          && request.resource.data.status == 'pending';
        allow update: if isOwner(workspaceId);
        allow delete: if isOwner(workspaceId) || (signedIn() && request.auth.uid == uid);
      }

      // -----------------------------------------------------------------
      // Pages — only Owner creates/deletes/manages permissions. Read/write
      // of an individual page and its rows is gated by allowedUsers.
      // -----------------------------------------------------------------
      match /pages/{pageId} {
        allow read: if isMember(workspaceId) && canAccessPage(workspaceId, resource.data);
        allow create: if isOwner(workspaceId) && request.resource.data.workspaceId == workspaceId;
        allow update, delete: if isOwner(workspaceId);

        // The person assigned as this page's "responsible" gets full control
        // over the page itself (rename, colour/icon, allowedUsers, columns)
        // — everything except reassigning who's responsible, which stays an
        // Owner/Admin-only action (see the Admin rule right below). This is
        // deliberately a plain per-field diff check, not a read/list rule,
        // so it carries none of the "list query" safety constraints that
        // apply to canAccessPage/the pages list query elsewhere in this file.
        allow update: if isMember(workspaceId)
          && resource.data.responsibleUserId == request.auth.uid
          && !('responsibleUserId' in request.resource.data.diff(resource.data).affectedKeys());

        // Admin (not Owner, who already has full rights above) may assign
        // or reassign who's responsible for this page — and only that field.
        allow update: if isMember(workspaceId)
          && myRole(workspaceId) == 'admin'
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['responsibleUserId', 'updatedAt']);

        match /rows/{rowId} {
          allow read: if isMember(workspaceId) &&
            canAccessPage(workspaceId, get(/databases/$(database)/documents/workspaces/$(workspaceId)/pages/$(pageId)).data);
          allow create, update, delete: if isMember(workspaceId) &&
            canEditPage(workspaceId, get(/databases/$(database)/documents/workspaces/$(workspaceId)/pages/$(pageId)).data);

          // Row-level comment threads — anyone with VIEW access to the
          // page (even a Viewer) can read and post; editing/deleting a
          // comment stays restricted to its own author.
          match /comments/{commentId} {
            allow read: if isMember(workspaceId) &&
              canAccessPage(workspaceId, get(/databases/$(database)/documents/workspaces/$(workspaceId)/pages/$(pageId)).data);
            allow create: if isMember(workspaceId) &&
              canAccessPage(workspaceId, get(/databases/$(database)/documents/workspaces/$(workspaceId)/pages/$(pageId)).data) &&
              request.resource.data.authorUid == request.auth.uid;
            allow update, delete: if isMember(workspaceId) && resource.data.authorUid == request.auth.uid;
          }
        }

        // -----------------------------------------------------------------
        // Subpages (inner tabs, e.g. "Январь"/"Февраль" inside "Финка") —
        // purely additive nested tables. Access mirrors the parent page
        // exactly: same allowedUsers-gated read, same non-viewer-gated
        // write/create — there is no separate per-subpage permission list.
        // -----------------------------------------------------------------
        match /subpages/{subPageId} {
          allow read: if isMember(workspaceId) &&
            canAccessPage(workspaceId, get(/databases/$(database)/documents/workspaces/$(workspaceId)/pages/$(pageId)).data);
          allow create, update, delete: if isMember(workspaceId) &&
            canEditPage(workspaceId, get(/databases/$(database)/documents/workspaces/$(workspaceId)/pages/$(pageId)).data);

          match /rows/{rowId} {
            allow read: if isMember(workspaceId) &&
              canAccessPage(workspaceId, get(/databases/$(database)/documents/workspaces/$(workspaceId)/pages/$(pageId)).data);
            allow create, update, delete: if isMember(workspaceId) &&
              canEditPage(workspaceId, get(/databases/$(database)/documents/workspaces/$(workspaceId)/pages/$(pageId)).data);
          }
        }

        // -----------------------------------------------------------------
        // Per-page chat — anyone with view access to this page (even a
        // Viewer) can read and post; editing/deleting a message stays
        // restricted to its own author.
        // -----------------------------------------------------------------
        match /chat/{messageId} {
          allow read: if isMember(workspaceId) &&
            canAccessPage(workspaceId, get(/databases/$(database)/documents/workspaces/$(workspaceId)/pages/$(pageId)).data);
          allow create: if isMember(workspaceId) &&
            canAccessPage(workspaceId, get(/databases/$(database)/documents/workspaces/$(workspaceId)/pages/$(pageId)).data) &&
            request.resource.data.authorUid == request.auth.uid;
          allow update, delete: if isMember(workspaceId) && resource.data.authorUid == request.auth.uid;
        }
      }

      // -----------------------------------------------------------------
      // History / audit log — Owner-only visibility, matching "Owner sees
      // everything" (regular members can still generate entries by editing
      // pages they have access to, they just can't browse the full log).
      // -----------------------------------------------------------------
      match /history/{entryId} {
        allow read: if isOwner(workspaceId);
        allow create: if isMember(workspaceId) && myRole(workspaceId) != 'viewer';
        allow delete: if isOwner(workspaceId);
      }

      // -----------------------------------------------------------------
      // Announcements — any member may read; only Owner/Admin may post,
      // edit, pin, archive, or delete.
      // -----------------------------------------------------------------
      match /announcements/{announcementId} {
        allow read: if isMember(workspaceId);
        allow create: if isOwnerOrAdmin(workspaceId) && request.resource.data.authorUid == request.auth.uid;
        allow update, delete: if isOwnerOrAdmin(workspaceId);
      }

      // -----------------------------------------------------------------
      // Notifications — fan-out, one doc per recipient. Only the Owner may
      // send (create) notifications. Each person may only read/update
      // (mark-as-read) their own — never anyone else's — and may delete
      // their own to clear it from their list.
      // -----------------------------------------------------------------
      match /notifications/{notificationId} {
        allow read, delete: if signedIn() && resource.data.targetUid == request.auth.uid;
        allow update: if signedIn()
          && resource.data.targetUid == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['read']);
        // Owner-only BROADCAST sending (the Announcements "notify" picker)
        // is enforced by the client UI, not here — this rule's job is just
        // to stop someone from faking who a notification is "from". Any
        // member may create a notification for this reason: @mentions in
        // chat/comments need to let ANY member ping ANY other member, not
        // just the Owner.
        allow create: if isMember(workspaceId) && request.resource.data.fromUid == request.auth.uid;
      }

      // -----------------------------------------------------------------
      // Workspace group chat — any member may read/post; editing or
      // deleting a message is restricted to its own author.
      // -----------------------------------------------------------------
      match /workspaceChat/{messageId} {
        allow read: if isMember(workspaceId);
        allow create: if isMember(workspaceId) && request.resource.data.authorUid == request.auth.uid;
        allow update, delete: if isMember(workspaceId) && resource.data.authorUid == request.auth.uid;
      }

      // -----------------------------------------------------------------
      // Private (1:1) messages. chatId is a deterministic pairing of the
      // two participants' uids, e.g. "<uidA>_<uidB>" sorted — so anyone
      // can be verified as a legitimate participant just from the chatId
      // itself, without needing a separate membership document.
      // -----------------------------------------------------------------
      function isInPrivateChat(chatId) {
        return chatId.split('_')[0] == request.auth.uid || chatId.split('_')[1] == request.auth.uid;
      }
      match /privateChats/{chatId} {
        // Two ways to qualify: the chatId-derived check (fine for reading
        // one already-known conversation) OR — needed for the "list all my
        // conversations" query, which filters by
        // where("participants","array-contains",uid) — a check on the
        // `participants` FIELD itself, so Firestore can prove that query
        // safe the same way it proves the allowedUsers array-contains
        // pattern used for pages.
        allow read, write: if isMember(workspaceId) &&
          (isInPrivateChat(chatId) || request.auth.uid in resource.data.get('participants', []));
        allow create: if isMember(workspaceId) &&
          (isInPrivateChat(chatId) || request.auth.uid in request.resource.data.get('participants', []));
        match /messages/{messageId} {
          allow read: if isMember(workspaceId) && isInPrivateChat(chatId);
          allow create: if isMember(workspaceId) && isInPrivateChat(chatId) && request.resource.data.authorUid == request.auth.uid;
          allow update, delete: if isMember(workspaceId) && isInPrivateChat(chatId) && resource.data.authorUid == request.auth.uid;
        }
      }

      // -----------------------------------------------------------------
      // Read markers — "when did I last read this conversation" per
      // context (workspace chat / a specific private chat). Strictly
      // self-service: the doc's own `uid` field must match the caller, and
      // list queries always filter by `where("uid","==",me)`, matching
      // this rule's per-doc condition exactly (the same safe pattern used
      // for notifications).
      // -----------------------------------------------------------------
      match /readMarkers/{markerId} {
        allow read: if isMember(workspaceId) && resource.data.uid == request.auth.uid;
        allow create, update: if isMember(workspaceId) && request.resource.data.uid == request.auth.uid;
      }
    }
  }
}
